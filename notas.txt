1. 
cd /workspaces/labsisop-buildroot

output/host/bin/i686-buildroot-linux-gnu-gcc -O2 -pthread sched_profiler.c -o sched_profiler

e2cp -p sched_profiler output/images/rootfs.ext2:/root/sched_profiler


2. Executar e gravar os traces
./output/images/start-qemu.sh serial-only


Dentro do QEMU:
cd /root

trace-cmd record -o trace_fifo.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_FIFO > saida_fifo.txt"

trace-cmd record -o trace_rr.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_RR > saida_rr.txt"

trace-cmd record -o trace_idle.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_IDLE > saida_idle.txt"

trace-cmd record -o trace_low_idle.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_LOW_IDLE > saida_low_idle.txt"

trace-cmd record -o trace_misto.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 2 SCHED_LOW_IDLE,SCHED_IDLE > saida_misto.txt"

sync
poweroff


3. Copiar os arquivos
cd /workspaces/labsisop-buildroot

for p in fifo rr idle low_idle misto; do
  e2cp output/images/rootfs.ext2:/root/trace_$p.dat trace_$p.dat
  e2cp output/images/rootfs.ext2:/root/saida_$p.txt saida_$p.txt
done



5. KernelShark (5 prints)
curl -fsSL https://gist.githubusercontent.com/miguelxvr/3c123ef67611652af9d2e45eccb8fc0a/raw/ | bash -s -- start


cd /workspaces/labsisop-buildroot

DISPLAY=:99 kernelshark "$PWD/trace_fifo.dat" &
